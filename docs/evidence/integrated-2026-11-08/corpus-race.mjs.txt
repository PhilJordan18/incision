import { chromium, expect } from '/workspace/incision-final-integrated/node_modules/@playwright/test/index.mjs';
import pg from '/workspace/incision-final-integrated/node_modules/pg/lib/index.js';
import assert from 'node:assert/strict';
const origin='http://localhost:3401';
const db=new pg.Client({connectionString:'postgresql://incision:incision_local_only@localhost:55432/incision'});
await db.connect();
const browser=await chromium.launch({headless:true});
try {
 const context=await browser.newContext({locale:'en-US',viewport:{width:1440,height:1000},reducedMotion:'reduce'});
 const page=await context.newPage();
 await page.goto(origin+'/sign-in');
 await page.locator('input[autocomplete="username"]').fill('demo-alice');
 await page.locator('input[type="password"]').fill('brume-alice-4817');
 await page.getByRole('button',{name:'Sign in',exact:true}).click();
 await expect(page).toHaveURL(/\/account$/);
 await page.goto(origin+'/rooms/new');
 await page.getByRole('button',{name:'Open the room',exact:true}).click();
 await expect(page.getByText('[.Live]',{exact:true})).toBeVisible();
 const code=page.url().split('/').at(-1);
 const settings=page.getByRole('region',{name:'Race settings'});
 await settings.getByRole('button',{name:'Edit settings'}).click();
 await settings.getByRole('combobox',{name:'Game mode',exact:true}).selectOption('arcade');
 await settings.getByRole('button',{name:'Save settings'}).click();
 await expect(settings.getByRole('status')).toHaveText('Settings saved.');
 const bots=page.getByRole('region',{name:'Bots',exact:true});
 await bots.getByRole('combobox').selectOption('impossible');
 await bots.getByRole('button',{name:'Add a bot',exact:true}).click();
 await expect(bots.getByRole('listitem')).toHaveCount(1);
 await page.getByRole('button',{name:'Start the race',exact:true}).click();
 const typing=page.getByLabel('Your typing',{exact:true});
 await expect(typing).toBeEnabled({timeout:10000});
 const inventory=page.locator('section:has(#race-bonuses-heading) ol').first();
 await expect(inventory.locator('li').first()).toBeVisible({timeout:40000});
 await typing.focus(); await page.keyboard.press('Enter');
 await expect(page.locator('section:has(#race-bonuses-heading) ol').nth(1).locator('li').first()).toBeVisible();
 await expect(inventory.locator('li')).toHaveCount(2,{timeout:40000});
 await page.getByRole('button',{name:'Give up the race',exact:true}).click();
 await page.getByRole('button',{name:'Yes, give up',exact:true}).click();
 await expect(page.getByRole('heading',{level:1})).toContainText('End of the',{timeout:40000});
 const journal=page.getByRole('region',{name:'Bonus journal',exact:true});
 const before=await journal.textContent();
 await page.reload(); await expect(journal).toHaveText(before);
 const result=await db.query('select r.id,r.config_snapshot from races r join lobbies l on l.current_race_id=r.id where l.code=$1',[code]);
 const race=result.rows[0]; assert(race); assert.equal(race.config_snapshot.corpusVersion,'2026-10-25.4');
 assert(!JSON.stringify(race.config_snapshot).includes('fixture'));
 const rows=(await db.query('select e.kind,r.outcome,r.bonus_journal from race_results r join race_entrants e on e.id=r.entrant_id where r.race_id=$1 order by e.ordinal',[race.id])).rows;
 assert.equal(rows.length,2);
 const human=rows.find(r=>r.kind==='human');
 assert.equal(human.bonus_journal.grants.filter(g=>g.activation).length,1);
 assert.equal(human.bonus_journal.grants.filter(g=>!g.activation).length,2);
 await page.goto(origin+'/history');
 await page.locator(`a[href="/races/${race.id}"]`).click();
 await expect(journal).toHaveText(before);
 await journal.locator('summary').first().click();
 await page.screenshot({path:'/workspace/scratch/integrated-2026-11-08/corpus-race.png',fullPage:true});
 console.log(JSON.stringify({result:'PASS',corpusVersion:race.config_snapshot.corpusVersion,participants:2,used:1,unused:2,outcomes:rows.map(r=>({kind:r.kind,outcome:r.outcome})),reloadAndHistoryIdentical:true,fixtureTexts:false,forcedBonusSeed:false},null,2));
} finally {await browser.close();await db.end();}
